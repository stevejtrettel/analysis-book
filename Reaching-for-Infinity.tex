\chapter{Foundations}\label{foundations}

\chapter{Sets and Functions}\label{sec-foundations-sets-functions}

{[}PROSE: Opening---before we can discuss numbers, we need the language
of sets and functions. This section establishes notation and vocabulary.
Most of this should be familiar from an introduction to proofs course;
we collect it here for reference.{]}

\section{Sets}\label{sets}

\begin{definition}[Set]\protect\hypertarget{def-foundations-set}{}\label{def-foundations-set}

A \emph{set} is a collection of objects, called \emph{elements} (or
\emph{members}).

\end{definition}

{[}PROSE: Notation. We write \(x \in A\) to mean \(x\) is an element of
\(A\), and \(x \notin A\) to mean \(x\) is not an element of \(A\).{]}

{[}PROSE: Specifying sets. A set can be described by listing its
elements: \(\{1, 2, 3\}\). For larger or infinite sets, we use
set-builder notation: \(\{x : P(x)\}\) is the set of all \(x\)
satisfying property \(P\).{]}

\begin{definition}[Standard Number
Sets]\protect\hypertarget{def-foundations-standard-sets}{}\label{def-foundations-standard-sets}

~

\begin{itemize}
\tightlist
\item
  \(\NN = \{1, 2, 3, \ldots\}\) --- the \emph{natural numbers}
\item
  \(\ZZ = \{\ldots, -2, -1, 0, 1, 2, \ldots\}\) --- the \emph{integers}
\item
  \(\QQ = \{p/q : p, q \in \ZZ,\, q \neq 0\}\) --- the \emph{rational
  numbers}
\end{itemize}

\end{definition}

{[}PROSE: Note that our convention is \(0 \notin \NN\). Some texts
include \(0\) in \(\NN\); one must check conventions.{]}

\begin{definition}[Subset]\protect\hypertarget{def-foundations-subset}{}\label{def-foundations-subset}

\(A \subseteq B\) means every element of \(A\) is an element of \(B\):
\[A \subseteq B \iff \forall x,\, (x \in A \implies x \in B)\]

\end{definition}

\begin{definition}[Set
Equality]\protect\hypertarget{def-foundations-set-equality}{}\label{def-foundations-set-equality}

\(A = B\) means \(A \subseteq B\) and \(B \subseteq A\).

\end{definition}

{[}PROSE: To prove two sets are equal, we typically prove containment
both ways.{]}

\section{Set Operations}\label{set-operations}

\begin{definition}[Union, Intersection,
Difference]\protect\hypertarget{def-foundations-set-operations}{}\label{def-foundations-set-operations}

~

\begin{itemize}
\tightlist
\item
  \emph{Union}: \(A \cup B = \{x : x \in A \text{ or } x \in B\}\)
\item
  \emph{Intersection}:
  \(A \cap B = \{x : x \in A \text{ and } x \in B\}\)
\item
  \emph{Difference}:
  \(A \setminus B = \{x : x \in A \text{ and } x \notin B\}\)
\end{itemize}

\end{definition}

\begin{definition}[Cartesian
Product]\protect\hypertarget{def-foundations-cartesian-product}{}\label{def-foundations-cartesian-product}

\(A \times B = \{(a, b) : a \in A,\, b \in B\}\)

\end{definition}

{[}PROSE: The Cartesian product consists of ordered pairs. Note
\((a, b) \neq (b, a)\) unless \(a = b\).{]}

{[}INLINE EXERCISE: Prove \((A \cup B)^c = A^c \cap B^c\) where \(A^c\)
denotes the complement of \(A\). (De Morgan's Law){]}

\section{Functions}\label{functions}

\begin{definition}[Function]\protect\hypertarget{def-foundations-function}{}\label{def-foundations-function}

A \emph{function} \(f: A \to B\) is a rule that assigns to each element
of \(A\) exactly one element of \(B\).

\begin{itemize}
\tightlist
\item
  \(A\) is the \emph{domain}
\item
  \(B\) is the \emph{codomain}
\item
  \(f(A) = \{f(x) : x \in A\}\) is the \emph{range} (or \emph{image})
\end{itemize}

\end{definition}

{[}PROSE: Important: functions are assignments, not formulas. The rule
``f(x) = 1 if x is rational, 0 if x is irrational'' defines a perfectly
good function ℝ → \{0, 1\}, even though there's no algebraic formula.{]}

{[}PROSE: Notation. We write \(f: A \to B\) to indicate the domain and
codomain, and \(x \mapsto f(x)\) to indicate what \(f\) does to
elements. For example, \(f: \RR \to \RR\), \(x \mapsto x^2\).{]}

\section{Types of Functions}\label{types-of-functions}

\begin{definition}[Injective]\protect\hypertarget{def-foundations-injective}{}\label{def-foundations-injective}

A function \(f: A \to B\) is \emph{injective} (or \emph{one-to-one}) if
\[f(x_1) = f(x_2) \implies x_1 = x_2\]

\end{definition}

\begin{definition}[Surjective]\protect\hypertarget{def-foundations-surjective}{}\label{def-foundations-surjective}

A function \(f: A \to B\) is \emph{surjective} (or \emph{onto}) if for
every \(y \in B\), there exists \(x \in A\) with \(f(x) = y\).
Equivalently, \(f(A) = B\).

\end{definition}

\begin{definition}[Bijective]\protect\hypertarget{def-foundations-bijective}{}\label{def-foundations-bijective}

A function is \emph{bijective} if it is both injective and surjective.

\end{definition}

{[}PROSE: Examples. The function \(f: \RR \to \RR\), \(x \mapsto x^2\)
is neither injective (\(f(1) = f(-1)\)) nor surjective (\(-1\) is not in
the range). The function \(f: \RR \to \RR\), \(x \mapsto x^3\) is
bijective.{]}

\section{Operations on Functions}\label{operations-on-functions}

\begin{definition}[Composition]\protect\hypertarget{def-foundations-composition}{}\label{def-foundations-composition}

If \(f: A \to B\) and \(g: B \to C\), the \emph{composition}
\(g \circ f: A \to C\) is defined by \[(g \circ f)(x) = g(f(x))\]

\end{definition}

{[}PROSE: Note the order: \(g \circ f\) means ``first \(f\), then
\(g\).''{]}

\begin{definition}[Inverse
Function]\protect\hypertarget{def-foundations-inverse}{}\label{def-foundations-inverse}

If \(f: A \to B\) is bijective, the \emph{inverse} \(f^{-1}: B \to A\)
is defined by \(f^{-1}(y) = x\) where \(f(x) = y\).

\end{definition}

\begin{proposition}[Inverses and
Bijections]\protect\hypertarget{prp-foundations-inverse-iff-bijective}{}\label{prp-foundations-inverse-iff-bijective}

A function \(f: A \to B\) has an inverse if and only if \(f\) is
bijective.

\end{proposition}

{[}PROSE: Proof outline. If \(f\) has an inverse, then \(f\) is
injective (if \(f(x_1) = f(x_2)\), apply \(f^{-1}\) to both sides) and
surjective (for any \(y \in B\), we have \(f(f^{-1}(y)) = y\)).
Conversely, if \(f\) is bijective, then for each \(y \in B\) there
exists a unique \(x \in A\) with \(f(x) = y\), so defining
\(f^{-1}(y) = x\) gives a well-defined function.{]}

\section{Exercises}\label{exercises}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  Prove \((A \cap B)^c = A^c \cup B^c\). (De Morgan's Law)
\item
  Prove
  \(A \setminus (B \cup C) = (A \setminus B) \cap (A \setminus C)\).
\item
  Prove that composition of functions is associative: if \(f: A \to B\),
  \(g: B \to C\), and \(h: C \to D\), then
  \(h \circ (g \circ f) = (h \circ g) \circ f\).
\item
  Let \(f: A \to B\) be bijective. Prove that \(f^{-1}\) is also
  bijective, and \((f^{-1})^{-1} = f\).
\item
  Let \(f: A \to B\) and \(g: B \to C\) be bijective. Prove that
  \(g \circ f\) is bijective, and
  \((g \circ f)^{-1} = f^{-1} \circ g^{-1}\).
\item
  Prove: \(f: A \to B\) is bijective if and only if there exists
  \(g: B \to A\) with \(g \circ f = \mathrm{id}_A\) and
  \(f \circ g = \mathrm{id}_B\).
\item
  Let \(f: A \to B\) and \(g: B \to C\).

  \begin{itemize}
  \tightlist
  \item
    If \(g \circ f\) is injective, must \(f\) be injective? Must \(g\)?
  \item
    If \(g \circ f\) is surjective, must \(f\) be surjective? Must
    \(g\)?
  \end{itemize}
\item
  \textbf{(★)} Prove that for any set \(A\), there is no surjection
  \(f: A \to \mathcal{P}(A)\), where \(\mathcal{P}(A)\) is the power set
  of \(A\) (the set of all subsets of \(A\)).

  \emph{Hint: Consider the set \(S = \{a \in A : a \notin f(a)\}\).}
\end{enumerate}

\chapter{Ordered Fields}\label{sec-foundations-ordered-fields}

{[}PROSE: Opening---we axiomatize arithmetic abstractly.{]}

\section{The Field Axioms}\label{the-field-axioms}

\begin{definition}[Field]\protect\hypertarget{def-foundations-field}{}\label{def-foundations-field}

A \emph{field} is a set \(\FF\) with two binary operations, addition
(\(+\)) and multiplication (\(\cdot\)), satisfying:

\textbf{Commutativity.} \(a + b = b + a\) and \(a \cdot b = b \cdot a\)

\textbf{Associativity.} \((a + b) + c = a + (b + c)\) and
\((a \cdot b) \cdot c = a \cdot (b \cdot c)\)

\textbf{Identities.} There exist distinct elements \(0, 1 \in \FF\) with
\(a + 0 = a\) and \(a \cdot 1 = a\)

\textbf{Inverses.} For each \(a\), there exists \(-a\) with
\(a + (-a) = 0\). For each \(a \neq 0\), there exists \(a^{-1}\) with
\(a \cdot a^{-1} = 1\)

\textbf{Distributivity.} \(a \cdot (b + c) = a \cdot b + a \cdot c\)

\end{definition}

{[}PROSE: Example ℚ. Non-example ℤ. Notation conventions. Numerals via
\(2 = 1+1\), etc., and base-10.{]}

{[}INLINE EXERCISE: Prove \(2 + 2 = 4\).{]}

\section{Consequences of the Field
Axioms}\label{consequences-of-the-field-axioms}

\begin{proposition}[Uniqueness of
Inverses]\protect\hypertarget{prp-foundations-uniqueness-inverses}{}\label{prp-foundations-uniqueness-inverses}

Additive and multiplicative inverses are unique.

\end{proposition}

{[}PROSE: Corollaries \(-(-a) = a\) and \((a^{-1})^{-1} = a\).{]}

\begin{proposition}[Multiplication by
Zero]\protect\hypertarget{prp-foundations-zero-multiplication}{}\label{prp-foundations-zero-multiplication}

\(0 \cdot a = 0\) for all \(a\).

\end{proposition}

\begin{proposition}[Negation by
Multiplication]\protect\hypertarget{prp-foundations-negation}{}\label{prp-foundations-negation}

\((-1) \cdot a = -a\) for all \(a\).

\end{proposition}

{[}PROSE: Corollaries \((-1)(-1) = 1\) and \((-a)(-b) = ab\).{]}

{[}INLINE EXERCISE: Prove \((a + b)(a - b) = a^2 - b^2\).{]}

\section{The Positivity Axioms}\label{the-positivity-axioms}

\begin{definition}[Ordered
Field]\protect\hypertarget{def-foundations-ordered-field}{}\label{def-foundations-ordered-field}

An \emph{ordered field} is a field \(\FF\) with a distinguished subset
\(P \subseteq \FF\) (the \emph{positive elements}) satisfying:

\textbf{Closure under addition.} If \(a, b \in P\), then \(a + b \in P\)

\textbf{Closure under multiplication.} If \(a, b \in P\), then
\(a \cdot b \in P\)

\textbf{Trichotomy.} For each \(a \in \FF\), exactly one holds:
\(a \in P\), or \(a = 0\), or \(-a \in P\)

\end{definition}

\begin{definition}[Inequality]\protect\hypertarget{def-foundations-inequality}{}\label{def-foundations-inequality}

For \(a, b \in \FF\): we write \(a < b\) to mean \(b - a \in P\), and
\(a \leq b\) to mean \(a < b\) or \(a = b\).

\end{definition}

{[}PROSE: Example ℚ is an ordered field.{]}

\section{Properties of Order}\label{properties-of-order}

\begin{proposition}[Squares are
Nonnegative]\protect\hypertarget{prp-foundations-squares-nonnegative}{}\label{prp-foundations-squares-nonnegative}

\(a^2 \geq 0\) for all \(a\), with equality iff \(a = 0\).

\end{proposition}

\begin{corollary}[One is
Positive]\protect\hypertarget{cor-foundations-one-positive}{}\label{cor-foundations-one-positive}

\(1 > 0\).

\end{corollary}

\begin{proposition}[Transitivity]\protect\hypertarget{prp-foundations-transitivity}{}\label{prp-foundations-transitivity}

If \(a < b\) and \(b < c\), then \(a < c\).

\end{proposition}

\begin{proposition}[Inequalities and
Addition]\protect\hypertarget{prp-foundations-inequality-addition}{}\label{prp-foundations-inequality-addition}

If \(a < b\), then \(a + c < b + c\).

\end{proposition}

\begin{proposition}[Inequalities and
Multiplication]\protect\hypertarget{prp-foundations-inequality-multiplication}{}\label{prp-foundations-inequality-multiplication}

If \(a < b\) and \(c > 0\), then \(ac < bc\). If \(a < b\) and
\(c < 0\), then \(ac > bc\).

\end{proposition}

{[}PROSE: This explains why multiplying by negatives reverses
inequalities.{]}

{[}INLINE EXERCISE: If \(0 < a < b\), prove \(a^2 < b^2\).{]}

\section{Absolute Value}\label{absolute-value}

\begin{definition}[Absolute
Value]\protect\hypertarget{def-foundations-absolute-value}{}\label{def-foundations-absolute-value}

\[|a| = \begin{cases} a & \text{if } a \geq 0 \\ -a & \text{if } a < 0 \end{cases}\]

\end{definition}

{[}PROSE: Properties \(|a| \geq 0\), \(|ab| = |a||b|\). Geometric
interpretation: \(|a - b|\) is the distance between \(a\) and \(b\) on
the number line.{]}

\begin{proposition}[Triangle
Inequality]\protect\hypertarget{prp-foundations-triangle-inequality}{}\label{prp-foundations-triangle-inequality}

\(|a + b| \leq |a| + |b|\)

\end{proposition}

\begin{corollary}[Reverse Triangle
Inequality]\protect\hypertarget{cor-foundations-reverse-triangle}{}\label{cor-foundations-reverse-triangle}

\(\bigl||a| - |b|\bigr| \leq |a - b|\)

\end{corollary}

{[}INLINE EXERCISE: Prove
\(|a_1 + \cdots + a_n| \leq |a_1| + \cdots + |a_n|\) by induction.{]}

\section{Intervals}\label{intervals}

{[}PROSE: With order and distance in hand, we can describe subsets of
the number line by specifying which points lie between two endpoints.{]}

\begin{definition}[Intervals]\protect\hypertarget{def-foundations-intervals}{}\label{def-foundations-intervals}

For \(a, b \in \FF\) with \(a < b\):

\begin{itemize}
\tightlist
\item
  \((a, b) = \{x \in \FF : a < x < b\}\) --- \emph{open interval}
\item
  \([a, b] = \{x \in \FF : a \leq x \leq b\}\) --- \emph{closed
  interval}
\item
  \([a, b) = \{x \in \FF : a \leq x < b\}\) --- \emph{half-open}
\item
  \((a, b] = \{x \in \FF : a < x \leq b\}\) --- \emph{half-open}
\end{itemize}

\end{definition}

\begin{refremark}[Intervals and Absolute Value]
The open interval centered at \(a\) with radius \(r > 0\) can be written
\[(a - r, a + r) = \{x \in \FF : |x - a| < r\}\] Similarly,
\([a - r, a + r] = \{x : |x - a| \leq r\}\).

\label{rem-foundations-interval-absolute-value}

\end{refremark}

{[}PROSE: For unbounded sets, we introduce symbols \(\infty\) and
\(-\infty\). These are \emph{not} elements of the field---they are
purely notational shorthand for ``unbounded.''{]}

\begin{definition}[Rays]\protect\hypertarget{def-foundations-rays}{}\label{def-foundations-rays}

For \(a \in \FF\):

\begin{itemize}
\tightlist
\item
  \([a, \infty) = \{x \in \FF : x \geq a\}\)
\item
  \((a, \infty) = \{x \in \FF : x > a\}\)
\item
  \((-\infty, a] = \{x \in \FF : x \leq a\}\)
\item
  \((-\infty, a) = \{x \in \FF : x < a\}\)
\item
  \((-\infty, \infty) = \FF\)
\end{itemize}

\end{definition}

\begin{refremark}[A Warning]
The symbols \(\infty\) and \(-\infty\) are convenient notation, but they
are not elements of any ordered field. Expressions like
``\(\infty + 1\)'' or ``\(\infty / \infty\)'' are meaningless in this
context.

\label{rem-foundations-infinity-not-number}

\end{refremark}

\subsection{The Intermediate Value Theorem
Fails}\label{the-intermediate-value-theorem-fails}

\begin{example}[IVT Fails in
ℚ]\protect\hypertarget{exm-foundations-ivt-fails}{}\label{exm-foundations-ivt-fails}

Let \(f(x) = x^2 - 2\) on \([1, 2] \cap \QQ\). Then \(f(1) = -1 < 0\)
and \(f(2) = 2 > 0\). If IVT held in ℚ, some \(c \in [1,2] \cap \QQ\)
would satisfy \(f(c) = 0\). But \(f(c) = 0\) means \(c^2 = 2\), which
has no rational solution.

\end{example}

{[}PROSE: IVT guarantees continuous functions hit every value between
endpoints. Foundation of existence arguments: roots, fixed points,
solutions to f(x)=y. Without IVT, cannot prove solutions exist.{]}

\(\sqrt{2} \notin \QQ\), this is defined and continuous at every
rational in \([1,2]\). Yet \(f\) is unbounded: as \(x \to \sqrt{2}\)
through rationals, \(|f(x)| \to \infty\).

\end{example}

{[}PROSE: The hole at √2 lets a continuous function escape to infinity.
EVT guarantees continuous functions on closed intervals achieve max/min.
Without EVT, optimization has no foundation.{]}

\section{Bounds}\label{bounds}

\begin{definition}[Bounds]\protect\hypertarget{def-foundations-bounds}{}\label{def-foundations-bounds}

Let \(S\) be a nonempty subset of an ordered field.

\begin{itemize}
\tightlist
\item
  \(M\) is an \emph{upper bound} for \(S\) if \(x \leq M\) for all
  \(x \in S\)
\item
  \(S\) is \emph{bounded above} if an upper bound exists
\item
  \(m\) is a \emph{lower bound} for \(S\) if \(m \leq x\) for all
  \(x \in S\)
\item
  \(S\) is \emph{bounded below} if a lower bound exists
\item
  \(S\) is \emph{bounded} if it is both bounded above and below
\end{itemize}

\end{definition}

{[}PROSE: Examples. The interval (0, 1) is bounded above by 1, also by
2, by 47, etc. Many upper bounds exist. The natural numbers ℕ have lower
bounds (0, −5, etc.) but no upper bound.{]}

\section{Supremum and Infimum}\label{supremum-and-infimum}

{[}PROSE: Among all upper bounds, we want the \emph{best} one---the
smallest. This may or may not be in the set.{]}

\begin{definition}[Supremum]\protect\hypertarget{def-foundations-supremum}{}\label{def-foundations-supremum}

Let \(S\) be a nonempty set that is bounded above. The \emph{supremum}
of \(S\), written \(\sup S\), is the \emph{least upper bound}: a number
\(\sigma\) such that

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \(\sigma\) is an upper bound for \(S\)
\item
  If \(M\) is any upper bound for \(S\), then \(\sigma \leq M\)
\end{enumerate}

\end{definition}

\begin{definition}[Infimum]\protect\hypertarget{def-foundations-infimum}{}\label{def-foundations-infimum}

Let \(S\) be a nonempty set that is bounded below. The \emph{infimum} of
\(S\), written \(\inf S\), is the \emph{greatest lower bound}: a number
\(\lambda\) such that

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \(\lambda\) is a lower bound for \(S\)
\item
  If \(m\) is any lower bound for \(S\), then \(m \leq \lambda\)
\end{enumerate}

\end{definition}

\begin{example}[Supremum Not in the
Set]\protect\hypertarget{exm-foundations-sup-not-in-set}{}\label{exm-foundations-sup-not-in-set}

Let \(S = \{1 - 1/n : n \in \NN\} = \{0, 1/2, 2/3, 3/4, \ldots\}\). The
upper bounds for \(S\) are all numbers \(\geq 1\). The least of these is
\(1\), so \(\sup S = 1\). But \(1 \notin S\)---the supremum need not
belong to the set.

\end{example}

\begin{example}[Supremum in the
Set]\protect\hypertarget{exm-foundations-sup-in-set}{}\label{exm-foundations-sup-in-set}

Let \(S = [0, 1] = \{x : 0 \leq x \leq 1\}\). Then \(\sup S = 1\) and
\(1 \in S\). When the supremum is in the set, it is also the maximum.

\end{example}

\begin{example}[Open
Interval]\protect\hypertarget{exm-foundations-open-interval-sup}{}\label{exm-foundations-open-interval-sup}

Let \(S = (0, 1)\). Then \(\sup S = 1\) and \(\inf S = 0\), but neither
belongs to \(S\).

\end{example}

\begin{example}[No Supremum in
ℚ]\protect\hypertarget{exm-foundations-no-sup-in-Q}{}\label{exm-foundations-no-sup-in-Q}

Let \(S = \{x \in \QQ : x > 0,\, x^2 < 2\}\). This set is bounded above
in \(\QQ\)---for instance, \(2\) is an upper bound. But \(S\) has no
supremum in \(\QQ\): any rational upper bound \(M\) satisfies
\(M^2 \geq 2\), and if \(M^2 > 2\) we can find a smaller upper bound.
The least upper bound would satisfy \(M^2 = 2\), which has no rational
solution.

\end{example}

{[}PROSE: This example shows precisely how ``holes'' in ℚ manifest: as
bounded sets without suprema.{]}

\begin{proposition}[Uniqueness]\protect\hypertarget{prp-foundations-supremum-unique}{}\label{prp-foundations-supremum-unique}

If \(\sup S\) exists, it is unique. Similarly for \(\inf S\).

\end{proposition}

{[}INLINE EXERCISE: Prove uniqueness of the supremum. Hint: if
\(\sigma\) and \(\sigma'\) are both least upper bounds, show
\(\sigma \leq \sigma'\) and \(\sigma' \leq \sigma\).{]}

\end{definition}

\section{The ε-Characterization}\label{the-ux3b5-characterization}

{[}PROSE: How do we actually use suprema in proofs? The key technique:
``get within ε of the sup.''{]}

\begin{proposition}[ε-Characterization of
Supremum]\protect\hypertarget{prp-foundations-epsilon-sup}{}\label{prp-foundations-epsilon-sup}

Let \(S\) be nonempty and bounded above. Then \(\sigma = \sup S\) if and
only if:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \(\sigma\) is an upper bound for \(S\)
\item
  For every \(\varepsilon > 0\), there exists \(x \in S\) with
  \(x > \sigma - \varepsilon\)
\end{enumerate}

\end{proposition}

{[}PROSE: Interpretation: condition 2 says ``nothing smaller than σ is
an upper bound''---for any ε, we can find an element of S within ε of σ.
This is the workhorse for supremum arguments.{]}

\begin{proposition}[ε-Characterization of
Infimum]\protect\hypertarget{prp-foundations-epsilon-inf}{}\label{prp-foundations-epsilon-inf}

Let \(S\) be nonempty and bounded below. Then \(\lambda = \inf S\) if
and only if:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \(\lambda\) is a lower bound for \(S\)
\item
  For every \(\varepsilon > 0\), there exists \(x \in S\) with
  \(x < \lambda + \varepsilon\)
\end{enumerate}

\end{proposition}

{[}PROSE: State without repeating the proof---symmetric argument.{]}

\section{Working with Suprema and
Infima}\label{working-with-suprema-and-infima}

\begin{example}[Translation]\protect\hypertarget{exm-foundations-sup-translation}{}\label{exm-foundations-sup-translation}

Let \(A\) be a bounded set and \(c \in \FF\). Define
\(A + c = \{a + c : a \in A\}\). Then \[\sup(A + c) = \sup A + c\]

\end{example}

{[}PROSE: Proof has two parts: (1) show \(\sup A + c\) is an upper bound
for \(A + c\), (2) show it's the least. Work through both steps
explicitly---this is a model for how sup arguments go.{]}

\begin{example}[Scaling by a Positive
Constant]\protect\hypertarget{exm-foundations-sup-scaling}{}\label{exm-foundations-sup-scaling}

Let \(A\) be a bounded set and \(c > 0\). Define
\(cA = \{ca : a \in A\}\). Then \[\sup(cA) = c \cdot \sup A\]

\end{example}

{[}PROSE: Similar two-part structure. Emphasize: this fails for
\(c < 0\) (sup and inf swap).{]}

\begin{proposition}[Subsets Have Smaller
Suprema]\protect\hypertarget{prp-foundations-sup-subset}{}\label{prp-foundations-sup-subset}

If \(A \subseteq B\) and both are nonempty and bounded above, then
\(\sup A \leq \sup B\).

\end{proposition}

{[}PROSE: Key insight: \(\sup B\) is an upper bound for \(B\), hence for
\(A\), hence \(\geq \sup A\).{]}

\begin{proposition}[Supremum of a
Union]\protect\hypertarget{prp-foundations-sup-union}{}\label{prp-foundations-sup-union}

If \(A\) and \(B\) are nonempty and bounded above, then
\[\sup(A \cup B) = \max\{\sup A, \sup B\}\]

\end{proposition}

\section{The Nested Interval
Property}\label{the-nested-interval-property}

{[}PROSE: Now we prove that completeness delivers what we wanted: in a
complete ordered field, nested intervals have nonempty intersection.{]}

\begin{definition}[Nested
Intervals]\protect\hypertarget{def-foundations-nested-intervals}{}\label{def-foundations-nested-intervals}

A sequence of intervals \(I_1, I_2, I_3, \ldots\) is \emph{nested} if
\(I_{n+1} \subseteq I_n\) for all \(n\).

\end{definition}

\begin{theorem}[Nested Interval
Property]\protect\hypertarget{thm-foundations-nested-intervals}{}\label{thm-foundations-nested-intervals}

Let \(\FF\) be a complete ordered field, and let
\([a_1, b_1] \supseteq [a_2, b_2] \supseteq [a_3, b_3] \supseteq \cdots\)
be a nested sequence of closed intervals in \(\FF\). Then
\[\bigcap_{n=1}^{\infty} [a_n, b_n] \neq \varnothing\]

\end{theorem}

{[}PROSE: Proof outline: Let \(A = \{a_1, a_2, a_3, \ldots\}\). This is
nonempty and bounded above by any \(b_n\) (since intervals are nested).
By completeness, \(\alpha = \sup A\) exists. Show
\(\alpha \in [a_n, b_n]\) for all \(n\): we have \(a_n \leq \alpha\)
(since \(\alpha\) is an upper bound for \(A\)) and \(\alpha \leq b_n\)
(since \(b_n\) is an upper bound for \(A\), so the least upper bound
satisfies \(\alpha \leq b_n\)).{]}

\begin{refremark}[When the Intersection is a Single Point]
If additionally \(\sup\{a_n\} = \inf\{b_n\}\)---for instance, if the
interval lengths \(b_n - a_n\) shrink to zero---then the intersection
contains exactly one point.

\label{rem-foundations-nested-single-point}

\end{refremark}

{[}PROSE: Closing paragraph---the nested interval property is exactly
what ancient mathematicians assumed when approximating π or √2.
Completeness makes it a theorem. In the next section, we take the bold
step of \emph{assuming} completeness as an axiom, defining the real
numbers as a complete ordered field, and exploring what this buys us.{]}

  \(x < \lambda + \varepsilon\).
\item
  Let \(A, B\) be nonempty bounded subsets of a complete field with
  \(A \subseteq B\). Prove that \(\sup A \leq \sup B\).
\item
  Let \(c > 0\) and \(A\) be a bounded set. Prove
  \(\sup(cA) = c \cdot \sup A\).
\item
  Let \(A, B\) be nonempty bounded sets. Prove:

  \begin{itemize}
  \tightlist
  \item
    \(\sup(A \cup B) = \max\{\sup A, \sup B\}\)
  \item
    \(\inf(A \cup B) = \min\{\inf A, \inf B\}\)
  \end{itemize}
\item
  For each set, find the supremum and infimum, or explain why they don't
  exist:

  \begin{itemize}
  \tightlist
  \item
    \([1, 3]\)
  \item
    \([1, 3)\)
  \item
    \(\{x \in \QQ : x^2 < 1\}\)
  \item
    \(\{x \in \QQ : x^3 < 1\}\)
  \item
    \(\{1 + 1/n : n \in \NN\}\)
  \item
    \(\{1 + (-1)^n/n : n \in \NN\}\)
  \end{itemize}
\item
  For each set, find the supremum and infimum, or explain why they don't
  exist:

  \begin{itemize}
  \tightlist
  \item
    \(\{(-1)^n/n : n \in \NN\}\)
  \item
    Fix \(\beta \in (0,1)\) and let \(B = \{\beta^n : n \in \NN\}\)
  \item
    Fix \(\gamma > 1\) and let \(C = \{\gamma^n : n \in \NN\}\)
  \end{itemize}
\item
  Let \(A, B\) be nonempty with \(\sup A < \sup B\). Prove there exists
  \(b \in B\) which is an upper bound for \(A\). Give an example showing
  this can fail if we only assume \(\sup A \leq \sup B\).
\item
  Let \(A, B\) be open intervals with \(\sup A = \inf B\). True or
  false: it is possible to add a single point to \(A \cup B\) so the
  result is an interval. Explain your reasoning.
\item
  Show that the nested interval theorem fails for open intervals: find a
  sequence of nested open intervals whose intersection is empty.
\item
  \textbf{(★)} Let \([a_n, b_n]\) be nested closed intervals and assume
  \(\sup\{a_n\} = \inf\{b_n\}\). Prove \(\bigcap_n [a_n, b_n]\) contains
  exactly one point.
\item
  \textbf{(★ Equivalents to Completeness)} Prove the following are
  equivalent:

  \begin{itemize}
  \tightlist
  \item
    \begin{enumerate}
    \def\labelenumii{(\arabic{enumii})}
    \tightlist
    \item
      Every nonempty set bounded above has a supremum
    \end{enumerate}
  \item
    \begin{enumerate}
    \def\labelenumii{(\arabic{enumii})}
    \setcounter{enumii}{1}
    \tightlist
    \item
      Every nonempty set bounded below has an infimum
    \end{enumerate}
  \end{itemize}

  \emph{Hint: For a set \(S\), consider \(-S = \{-s : s \in S\}\) and
  show \(\sup(-S) = -\inf(S)\).}
\end{enumerate}

\chapter{The Real Numbers}\label{sec-foundations-reals}

{[}PROSE: Opening---the bold move. We \emph{declare} the real numbers to
be a complete ordered field. This is the axiomatic method: we specify
properties, not construction. We don't say what real numbers \emph{are},
we say what they \emph{do}.{]}

\begin{definition}[The Real
Numbers]\protect\hypertarget{def-foundations-real-numbers}{}\label{def-foundations-real-numbers}

The \emph{real numbers} \(\RR\) are a complete ordered field.

\end{definition}

{[}PROSE: Historical context. This was Huntington's approach (1903). But
one might worry: do such axioms describe anything at all? (Easy to write
inconsistent axioms.) And if they do, do they pick out one thing or
many? (Group axioms describe a whole subject, not one example.){]}

\begin{theorem}[Existence and
Uniqueness]\protect\hypertarget{thm-foundations-reals-unique}{}\label{thm-foundations-reals-unique}

There exists a complete ordered field, and it is unique up to
isomorphism.

\end{theorem}

{[}PROSE: We won't prove this---existence requires a construction
(Dedekind cuts or Cauchy sequences), uniqueness requires showing any two
complete ordered fields are isomorphic. But the theorem is important:
our axioms completely characterize something. This is evidence we've
succeeded.{]}

\begin{refremark}[Conventions for Sup and Inf]
Using the notation from §1.2, we extend our sup and inf conventions: if
\(S\) is nonempty but not bounded above, we write \(\sup S = \infty\).
If \(S\) is nonempty but not bounded below, we write
\(\inf S = -\infty\). For the empty set, we adopt the convention
\(\sup \varnothing = -\infty\) and \(\inf \varnothing = \infty\).

\label{rem-foundations-sup-inf-infinity}

\end{refremark}

\section{No Infinite Numbers}\label{no-infinite-numbers}

{[}PROSE: What have we created? First, we check there are no
``monsters''---no infinite or infinitesimal numbers. The real line is
``normal.''{]}

\begin{theorem}[ℕ is
Unbounded]\protect\hypertarget{thm-foundations-N-unbounded}{}\label{thm-foundations-N-unbounded}

The natural numbers have no upper bound in \(\RR\). Equivalently: for
any \(x \in \RR\), there exists \(n \in \NN\) with \(n > x\).

\end{theorem}

{[}PROSE: Proof uses completeness: if ℕ were bounded, sup ℕ would exist.
Let s = sup ℕ. Then s - 1 is not an upper bound, so some n ∈ ℕ satisfies
n \textgreater{} s - 1. But then n + 1 \textgreater{} s, contradicting s
being an upper bound.{]}

\begin{corollary}[No
Infinitesimals]\protect\hypertarget{cor-foundations-no-infinitesimals}{}\label{cor-foundations-no-infinitesimals}

For any \(\varepsilon > 0\), there exists \(n \in \NN\) with
\(1/n < \varepsilon\).

\end{corollary}

{[}PROSE: Brief comment---people have wondered about formalizing
infinitesimal numbers (numbers smaller than every 1/n but positive). In
fact they don't exist in ℝ: if ε were infinitesimal, then 1/ε would be
infinite, contradicting the theorem above.{]}

\section{The Archimedean Property}\label{the-archimedean-property}

{[}PROSE: Repackage the above as the Archimedean property. Historical
note: Archimedes took this as an axiom in \emph{The Sphere and the
Cylinder}. Also appears in Euclid Book V. Mnemonic: ``you can empty the
ocean with a teaspoon''---no matter how small a is, enough copies na
will exceed any b.{]}

\begin{definition}[Archimedean
Field]\protect\hypertarget{def-foundations-archimedean}{}\label{def-foundations-archimedean}

An ordered field \(\FF\) is \emph{Archimedean} if for every positive
\(a, b \in \FF\), there exists \(n \in \NN\) with \(na > b\).

\end{definition}

\begin{theorem}[Complete Fields are
Archimedean]\protect\hypertarget{thm-foundations-reals-archimedean}{}\label{thm-foundations-reals-archimedean}

Every complete ordered field is Archimedean.

\end{theorem}

{[}PROSE: Proof: given positive a, b, the number b/a is finite (no
infinites), so some n \textgreater{} b/a, hence na \textgreater{} b.{]}

{[}INLINE EXERCISE: Prove ℚ is Archimedean directly from the definition
of rational numbers.{]}

\section{Density}\label{density}

\begin{theorem}[Density of
ℚ]\protect\hypertarget{thm-foundations-Q-dense}{}\label{thm-foundations-Q-dense}

Between any two real numbers lies a rational.

\end{theorem}

{[}PROSE: Proof outline: Let a \textless{} b. By Archimedean, choose n
with 1/n \textless{} b - a. By Archimedean again, the set \{m ∈ ℤ : m
\textgreater{} na\} is nonempty and bounded below. Let m be its least
element. Then a \textless{} m/n \textless{} b.{]}

\begin{theorem}[Density of
Irrationals]\protect\hypertarget{thm-foundations-irrationals-dense}{}\label{thm-foundations-irrationals-dense}

Between any two real numbers lies an irrational.

\end{theorem}

{[}PROSE: Proof: Let a \textless{} b. Choose rational r with a - √2
\textless{} r \textless{} b - √2. Then a \textless{} r + √2 \textless{}
b, and r + √2 is irrational.{]}

{[}PROSE: Interpretation---ℚ is everywhere dense in ℝ. There are no
``chunks'' of pure irrationals. We filled gaps, we didn't add alien
regions.{]}

\section{Roots Exist}\label{roots-exist}

{[}PROSE: The crisis of 1.3 was that √2 doesn't exist in ℚ. Now we prove
it exists in ℝ.{]}

\begin{theorem}[Existence of
√2]\protect\hypertarget{thm-foundations-sqrt2-exists}{}\label{thm-foundations-sqrt2-exists}

There exists a positive real number whose square is 2.

\end{theorem}

{[}PROSE: Full proof. Let S = \{x ∈ ℝ : x \textgreater{} 0, x²
\textless{} 2\}. S is nonempty (1 ∈ S) and bounded above (2 is an upper
bound). By completeness, s = sup S exists. We show s² = 2 by ruling out
s² \textless{} 2 and s² \textgreater{} 2.

\textbf{Case s² \textless{} 2}: Find h \textgreater{} 0 small enough
that (s+h)² \textless{} 2, so s+h ∈ S, contradicting s = sup S.

\textbf{Case s² \textgreater{} 2}: Find h \textgreater{} 0 small enough
that (s-h)² \textgreater{} 2, so s-h is still an upper bound,
contradicting s being least.

Work out the explicit bounds on h in each case.{]}

\begin{definition}[nth
Roots]\protect\hypertarget{def-foundations-nth-root}{}\label{def-foundations-nth-root}

For \(a \geq 0\) and \(n \in \NN\), define
\[\sqrt[n]{a} = \sup\{x \geq 0 : x^n < a\}\]

\end{definition}

\begin{theorem}[nth Roots
Exist]\protect\hypertarget{thm-foundations-nth-root-exists}{}\label{thm-foundations-nth-root-exists}

For \(a \geq 0\) and \(n \in \NN\), we have \((\sqrt[n]{a})^n = a\).

\end{theorem}

{[}PROSE: State theorem; proof similar to √2, details in exercises.{]}

\section{Rational and Real Exponents}\label{rational-and-real-exponents}

{[}PROSE: With roots in hand, we can define rational exponents, then
extend to all real exponents using completeness.{]}

\begin{definition}[Rational
Exponents]\protect\hypertarget{def-foundations-rational-exponents}{}\label{def-foundations-rational-exponents}

For \(a > 0\) and \(p/q \in \QQ\) with \(q > 0\), define
\[a^{p/q} = \left(\sqrt[q]{a}\right)^p\]

\end{definition}

{[}PROSE: This is well-defined (independent of representation of p/q)
and satisfies the expected laws: \(a^{r+s} = a^r a^s\),
\(a^{rs} = (a^r)^s\), and for \(a > 1\), \(r < s\) implies
\(a^r < a^s\).{]}

\begin{definition}[Real
Exponents]\protect\hypertarget{def-foundations-real-exponents}{}\label{def-foundations-real-exponents}

For \(a > 1\) and \(x \in \RR\), define
\[a^x = \sup\{a^r : r \in \QQ,\, r < x\}\] For \(0 < a < 1\), define
\(a^x = (1/a)^{-x}\).

\end{definition}

{[}PROSE: The functional equation \(a^{x+y} = a^x \cdot a^y\) holds for
all real x, y.{]}

\section{Logarithms}\label{logarithms}

\begin{definition}[Logarithms]\protect\hypertarget{def-foundations-logarithm}{}\label{def-foundations-logarithm}

For \(a > 0\), \(a \neq 1\), and \(y > 0\), define
\[\log_a(y) = \sup\{r \in \QQ : a^r < y\}\]

\end{definition}

{[}PROSE: This satisfies \(\log_a(xy) = \log_a(x) + \log_a(y)\), and
\(a^{\log_a(y)} = y\), and \(\log_a(a^x) = x\). So log and exp are
inverses.{]}

{[}PROSE: Interpretation---completeness lets us build all the functions
we need: roots, exponentials, logarithms. These are the ``new
treasures'' that ℝ provides.{]}

\section{Uncountability}\label{uncountability}

{[}PROSE: We've built ℝ to have no holes. It turns out to be
unimaginably vast---uncountably infinite.{]}

\begin{theorem}[ℝ is
Uncountable]\protect\hypertarget{thm-foundations-R-uncountable}{}\label{thm-foundations-R-uncountable}

The real numbers cannot be put in bijection with \(\NN\).

\end{theorem}

{[}PROSE: Proof via nested intervals. Suppose {[}0,1{]} is countable:
list elements as x₁, x₂, x₃, \ldots{} Build nested intervals {[}aₙ,
bₙ{]} avoiding xₙ at each stage, with lengths → 0. By nested interval
theorem, intersection contains some L. But L ≠ xₙ for all
n.~Contradiction.{]}

\begin{corollary}[Irrationals are
Uncountable]\protect\hypertarget{cor-foundations-irrationals-uncountable}{}\label{cor-foundations-irrationals-uncountable}

The set \(\RR \setminus \QQ\) of irrational numbers is uncountable.

\end{corollary}

{[}PROSE: Proof: ℝ = ℚ ∪ (ℝ ~ℚ). If irrationals were countable, ℝ would
be union of two countable sets, hence countable.{]}

\section{The Bigger Picture}\label{the-bigger-picture}

{[}PROSE: Since ℝ is uncountable but ℚ is countable, ``most'' real
numbers are irrational in a precise sense.{]}

{[}PROSE: More striking: the algebraic numbers 𝔸 are countable (there
are countably many polynomials, each with finitely many roots). So
transcendental numbers exist---in fact ``most'' reals are
transcendental!{]}

{[}PROSE: Even more striking: computable numbers (those whose digits can
be output by an algorithm) are countable. So ``most'' reals cannot be
computed. Definable numbers (those specifiable by a finite sentence) are
countable. So ``most'' reals cannot even be described.{]}

{[}PROSE: Closing---we built ℝ to fill the holes in ℚ. It turned out to
be an uncountable ocean in which the rationals, algebraics, and
computables are a countable dust.{]}

    \(x_i = 0.d_{i1}d_{i2}d_{i3}\ldots\)
  \item

    \(e_n \neq d_{nn}\) (and \(e_n \neq 0, 9\) to avoid issues with

  \end{enumerate}

  \tightlist

    bijection with \(\NN\).
  \end{enumerate}

  \emph{Hint: List pairs by diagonals: (1,1), (1,2), (2,1), (1,3),

  \begin{enumerate}
  \def\labelenumii{(\alph{enumii})}
  \setcounter{enumii}{1}
  \item
    Prove that a countable union of countable sets is countable.
  \item
    Conclude that \(\QQ\) is countable.
  \end{enumerate}
\item
  \textbf{(★★ Guided Exploration: Countability of Algebraic Numbers)}

  \begin{enumerate}
  \def\labelenumii{(\alph{enumii})}
  \item
    For each \(n \in \NN\), let \(A_n\) be the set of algebraic numbers
    that are roots of some polynomial of degree \(n\) with integer
    coefficients whose absolute values sum to at most \(n\). Prove
    \(A_n\) is finite.
  \item
    Prove that \(\AA = \bigcup_{n=1}^{\infty} A_n\).
  \item
    Conclude that the algebraic numbers are countable.
  \item
    Conclude that transcendental numbers exist (without constructing
    one!).
  \end{enumerate}
\end{enumerate}
