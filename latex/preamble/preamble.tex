% ============================================================================
% The book's print design. Hand-authored; everything derived from the
% environment registry (theorem declarations, cref names) is generated into
% main.tex by the compiler and does NOT live here.
%
% Engine: XeLaTeX + memoir (decided). Type: EB Garamond text with
% Garamond-Math — print math is genuinely Garamond — Plex Sans as the sans,
% JetBrains Mono for code, mirroring the site.
%
% Two switches, set before \input of this file or edited here:
%   \BookTrim     "letter" (wide-margin course notes, current) | "sixbynine"
%   \PlexHeads    define it for Plex Sans headings (site-matched);
%                 undefined = EB Garamond headings (classic, the default)
% ============================================================================

% ---- the heading face switch ----------------------------------------------
% \headfont: the face for chapter/section titles, theorem heads, captions.
% \kickerfont: small tracked-caps labels ("CHAPTER 1", running heads).

\ifdefined\PlexHeads
  % Plex has no small caps: kickers are tracked full caps (content must be
  % uppercase — use \MakeUppercase where marks supply mixed case).
  \newcommand\headfont{\plexsemibold}
  \newcommand\kickerfont{\plexmedium\addfontfeatures{LetterSpace=14}}
\else
  \newcommand\headfont{\garamondsemibold}
  \newcommand\kickerfont{\garamondsemibold\addfontfeatures{LetterSpace=14}\scshape}
\fi

\usepackage{xcolor}
\definecolor{inkfaint}{gray}{0.45} % kickers, numbers, notes — the only "color" in print

% ---- page geometry ---------------------------------------------------------

\def\trimletter{letter}
\ifx\BookTrim\trimletter
  % Letter stock, symmetric wide margins (course notes read on screen and
  % print at home — binding asymmetry would look odd on both).
  \setstocksize{11in}{8.5in}
  \settrimmedsize{11in}{8.5in}{*}
  % ~66 characters per line of Garamond at 12pt; classical vertical
  % weighting (more margin below the block than above, so text sits at the
  % visual center rather than the geometric one).
  \settypeblocksize{8.5in}{5.3in}{*}
  \setlrmargins{*}{*}{1}
  \setulmargins{*}{*}{1.4}
\else
  % 6x9 book block (the long-term trim), centered on its own stock.
  \setstocksize{9in}{6in}
  \settrimmedsize{9in}{6in}{*}
  \settypeblocksize{7.1in}{4.35in}{*}
  \setlrmargins{*}{*}{1}
  \setulmargins{*}{*}{1}
\fi

% ---- title page -------------------------------------------------------------
% Echoes the ratified web cover: kicker, large Garamond title, tracked-caps
% author, then the phase portrait.

\newcommand\BookTitlePage{%
  \begin{titlingpage}
    \centering
    \vspace*{1.3in}
    {\kickerfont\small\color{inkfaint}COURSE NOTES\par}
    \vspace{1.4\baselineskip}
    % Plex on purpose: in print, where serif title pages are the expected
    % default, the sans is the bold face — the mirror of the web cover's
    % Garamond-against-sans-chrome. (The author noticed the inversion.)
    {\headfont\fontsize{34}{40}\selectfont \BookTitle\par}
    % Subtitle and descriptor mirror the web cover: a second title line in
    % the body serif, then the small sans descriptor, then the author.
    \ifx\BookSubtitle\empty\else
      \vspace{0.5\baselineskip}
      {\normalfont\sffamily\itshape\fontsize{19}{24}\selectfont\color{inkfaint}\BookSubtitle\par}%
    \fi
    \ifx\BookDescriptor\empty\else
      \vspace{0.9\baselineskip}
      {\kickerfont\footnotesize\color{inkfaint}\BookDescriptor\par}%
    \fi
    \vspace{1.9\baselineskip}
    {\kickerfont\small \BookAuthor\par}
    \vspace{0.72in}
    \BookHeroFig% hero art from book.yml figure:, defined in main.tex (empty if unset)
    \vspace{0.72in}
    {\color{inkfaint}\parbox{4.1in}{\centering\BookBlurb}\par}
    \vfill
  \end{titlingpage}
}
